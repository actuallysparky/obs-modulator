cmake_minimum_required(VERSION 3.24)
project(obs-modulator VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_OBS_PLUGIN "Build the native OBS plugin" ON)
option(MODULATOR_DOCK "Build the optional OBS frontend monitor" ON)
option(MODULATOR_FLATPAK "Install into the OBS Flatpak extension layout (Linux only)" OFF)
if(MODULATOR_FLATPAK AND (NOT UNIX OR APPLE))
  message(FATAL_ERROR "MODULATOR_FLATPAK requires Linux")
endif()
option(BUILD_PLUGIN_TESTS "Build graphics-backed libobs integration tests" ON)
set(MODULATOR_OBS_DATA "" CACHE PATH "libobs effects/data directory for integration tests")
set(MODULATOR_GRAPHICS_MODULE "" CACHE FILEPATH "OBS graphics module for integration tests")
set(MODULATOR_SHADERFILTER "" CACHE FILEPATH "Optional Shaderfilter module to test")
set(MODULATOR_SHADERFILTER_DATA "" CACHE PATH "Shaderfilter resources")
set(MODULATOR_MAC_RPATH "" CACHE PATH "OBS framework directory for local test execution")
include(CTest)
function(modulator_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endfunction()
add_library(modulator-core STATIC src/lfo.cpp)
target_include_directories(modulator-core PUBLIC src)
set_target_properties(modulator-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
modulator_warnings(modulator-core)
if(BUILD_TESTING)
  add_executable(test-core tests/test-core.cpp)
  target_link_libraries(test-core PRIVATE modulator-core)
  add_test(NAME core COMMAND test-core)
  add_executable(test-history tests/test-history.cpp)
  target_include_directories(test-history PRIVATE src)
  modulator_warnings(test-history)
  add_test(NAME history COMMAND test-history)
endif()
if(BUILD_OBS_PLUGIN)
  find_package(libobs CONFIG REQUIRED)
  add_library(obs-modulator MODULE src/plugin.cpp src/monitor.cpp)
  target_link_libraries(obs-modulator PRIVATE modulator-core OBS::libobs)
  target_compile_definitions(obs-modulator PRIVATE MODULATOR_VERSION="${PROJECT_VERSION}")
  modulator_warnings(obs-modulator)
  set_target_properties(obs-modulator PROPERTIES PREFIX "")
  if(MODULATOR_DOCK)
    find_package(obs-frontend-api CONFIG REQUIRED HINTS "${libobs_DIR}")
    # OBS's installed SDK exports its real target name. The local Mac SDK
    # adapter retains the build-tree alias used by older preview builds.
    if(TARGET OBS::obs-frontend-api)
      set(modulator_frontend_target OBS::obs-frontend-api)
    elseif(TARGET OBS::frontend-api)
      set(modulator_frontend_target OBS::frontend-api)
    else()
      message(FATAL_ERROR "OBS frontend SDK did not export a supported target")
    endif()
    find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
    target_sources(obs-modulator PRIVATE src/dock.cpp src/frontend.cpp)
    target_compile_definitions(obs-modulator PRIVATE MODULATOR_DOCK)
    target_link_libraries(obs-modulator PRIVATE ${modulator_frontend_target} Qt6::Widgets)
  endif()
  if(BUILD_TESTING)
    if(BUILD_PLUGIN_TESTS)
      add_executable(test-monitor tests/test-monitor.cpp src/plugin.cpp src/monitor.cpp)
      target_include_directories(test-monitor PRIVATE src)
      target_compile_definitions(test-monitor PRIVATE MODULATOR_TESTING MODULATOR_VERSION="${PROJECT_VERSION}")
      target_link_libraries(test-monitor PRIVATE modulator-core OBS::libobs)
      modulator_warnings(test-monitor)
      if(APPLE)
        set_target_properties(test-monitor PROPERTIES BUILD_RPATH "${MODULATOR_MAC_RPATH}")
      endif()
    endif()
    if(MODULATOR_DOCK)
      add_executable(test-dock tests/test-dock.cpp src/dock.cpp src/monitor.cpp)
      target_include_directories(test-dock PRIVATE src)
      target_compile_definitions(test-dock PRIVATE MODULATOR_TESTING)
      target_link_libraries(test-dock PRIVATE Qt6::Widgets modulator-core)
      modulator_warnings(test-dock)
      add_test(NAME dock COMMAND test-dock)
      set_tests_properties(dock PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
      if(APPLE)
        get_filename_component(qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
        set_property(TEST dock APPEND PROPERTY ENVIRONMENT "QT_QPA_PLATFORM_PLUGIN_PATH=${qt_prefix}/plugins/platforms")
        set_target_properties(test-dock PROPERTIES BUILD_RPATH "${MODULATOR_MAC_RPATH}")
      endif()
    endif()
  endif()
  if(APPLE)
    set_target_properties(obs-modulator PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/data/Info.plist.in"
      BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
    add_custom_command(TARGET obs-modulator POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data/locale" "$<TARGET_BUNDLE_DIR:obs-modulator>/Contents/Resources/locale"
      COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:obs-modulator>")
    install(TARGETS obs-modulator LIBRARY DESTINATION .)
    # CMake may edit RPATH during installation, invalidating the build signature.
    install(CODE "execute_process(COMMAND codesign --force --sign - \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/obs-modulator.plugin\" COMMAND_ERROR_IS_FATAL ANY)")
  elseif(MODULATOR_FLATPAK)
    install(TARGETS obs-modulator LIBRARY DESTINATION lib/obs-plugins)
    install(DIRECTORY data/locale DESTINATION share/obs/obs-plugins/obs-modulator)
    install(FILES LICENSE DESTINATION share/licenses/obs-modulator)
    install(FILES packaging/flatpak/com.obsproject.Studio.Plugin.Modulator.metainfo.xml DESTINATION share/metainfo)
  else()
    install(TARGETS obs-modulator LIBRARY DESTINATION obs-modulator/bin/64bit RUNTIME DESTINATION obs-modulator/bin/64bit)
    install(DIRECTORY data/locale DESTINATION obs-modulator/data)
  endif()
  if(BUILD_TESTING AND BUILD_PLUGIN_TESTS)
    if(APPLE)
      enable_language(OBJCXX)
      set(platform_source tests/platform-macos.mm)
    elseif(UNIX)
      set(platform_source tests/platform-linux.cpp)
    else()
      message(FATAL_ERROR "Graphics integration adapter is not implemented for this platform; set BUILD_PLUGIN_TESTS=OFF explicitly")
    endif()
    add_executable(test-plugin tests/test-plugin.cpp ${platform_source})
    target_sources(test-monitor PRIVATE ${platform_source})
    if(APPLE)
      target_link_libraries(test-monitor PRIVATE "-framework AppKit")
    endif()
    add_test(NAME monitor COMMAND test-monitor "${MODULATOR_GRAPHICS_MODULE}" "${MODULATOR_OBS_DATA}")
    if(MODULATOR_DOCK)
      target_sources(test-monitor PRIVATE src/dock.cpp)
      target_compile_definitions(test-monitor PRIVATE MONITOR_UI_TEST)
      target_link_libraries(test-monitor PRIVATE Qt6::Widgets)
      set_tests_properties(monitor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
      if(APPLE)
        set_property(TEST monitor APPEND PROPERTY ENVIRONMENT "QT_QPA_PLATFORM_PLUGIN_PATH=${qt_prefix}/plugins/platforms")
      endif()
    endif()
    target_link_libraries(test-plugin PRIVATE OBS::libobs)
    if(APPLE)
      target_link_libraries(test-plugin PRIVATE "-framework AppKit")
      set_target_properties(test-plugin PROPERTIES BUILD_RPATH "${MODULATOR_MAC_RPATH}")
    endif()
    modulator_warnings(test-plugin)
    if(APPLE)
      set(plugin_data "$<TARGET_BUNDLE_DIR:obs-modulator>/Contents/Resources")
    else()
      set(plugin_data "${CMAKE_CURRENT_SOURCE_DIR}/data")
    endif()
    if(NOT MODULATOR_GRAPHICS_MODULE)
      message(FATAL_ERROR "Set MODULATOR_GRAPHICS_MODULE to the runtime graphics module")
    endif()
    set(integration_args $<TARGET_FILE:obs-modulator> "${plugin_data}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test.shader" "${MODULATOR_GRAPHICS_MODULE}" "${MODULATOR_OBS_DATA}")
    add_test(NAME plugin COMMAND test-plugin ${integration_args})
    set_tests_properties(plugin PROPERTIES TIMEOUT 90)
    if(MODULATOR_SHADERFILTER)
      if(NOT MODULATOR_SHADERFILTER_DATA)
        message(FATAL_ERROR "Shaderfilter test requires MODULATOR_SHADERFILTER_DATA")
      endif()
      add_test(NAME shaderfilter COMMAND test-plugin ${integration_args} "${MODULATOR_SHADERFILTER}" "${MODULATOR_SHADERFILTER_DATA}")
      set_tests_properties(shaderfilter PROPERTIES TIMEOUT 90)
    else()
      message(STATUS "Shaderfilter integration is not configured (synthetic plugin test remains enabled)")
    endif()
  endif()
endif()
